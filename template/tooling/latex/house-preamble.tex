% house-preamble.tex — the house LaTeX preamble for every business document (XeLaTeX).
%
% Every .tex document starts from skeleton.tex, which loads this file with
% \input{house-preamble}; `make pdf FILE=…` puts tooling/latex/ and assets/ on TEXINPUTS
% so it is found from any folder (after LOGO_DIRS, when tooling/project.mk sets it, so a
% logo is taken from the folder named there first). Template-owned: `copier update` merges
% improvements here.
%
% THE BRAND IS NOT SET HERE. Colours and fonts below are neutral defaults. The author's
% own values live in house-brand.tex (an optional file the author creates in
% tooling/latex/, never shipped by the template), loaded at the end of this file, so a
% template update never touches the brand. The brand guide maps each name (in the brand folder
% 00-project.md ## Paths names; by default standards/brand/brand-guide.md).
% house-brand.tex holds only \definecolor{house…}{HTML}{…} lines and font settings
% (\setmainfont{…}, \setfontfamily\headingfont{…}), each naming an installed font.
%
% Macros:
%   \dnote{…}                     drafting note: muted, bracketed. One whose text opens with
%                                  AUTHOR TO CONFIRM or VERIFY and a colon is a flag that
%                                  `make flags` lists; none may survive into an issued document.
%   \fillme                        a field the author must complete by hand:
%                                  prints [AWAITING USER INPUT].
%   clause environment             auto-numbered clauses 1. / 1.1 / (a); \label{cl:slug}
%                                  then clause~\ref{cl:slug}. Render twice so \ref resolves.
%   \ctitle{…}                     the title of a level-1 clause.
%   \ins{…} \del{…} \cmt{…}        redline: inserted, deleted, commented text.
%   \srule                         a short centred divider.
%   \housetitle{title}{name}{date} the title block.
%   housenotice environment        the disclaimer block (wording from the disclaimers file
%                                  00-project.md ## Paths names).
%   housecontrol environment       the document-control block: one `Label & Value \\` per row.
%   \houseclassification{level}    the classification, in the running header and footer of
%                                  every page (plain pages too). Call it in the preamble, after
%                                  \input{house-preamble}. A document that never calls it keeps
%                                  LaTeX's own page style: a page number at the foot, no header.

% ── House macros ─────────────────────────────────────────────────────────────────────
\newcommand{\dnote}[1]{{\color{housemuted}\small\itshape [#1]}}
\newcommand{\fillme}{\textbf{\textcolor{chgred}{[AWAITING USER INPUT]}}}
\newcommand{\ins}[1]{\textcolor{chgred}{#1}}
\newcommand{\del}[1]{\textcolor{chgred}{\sout{#1}}}
\newcommand{\cmt}[1]{{\small\leavevmode\textcolor{cmtblue}{#1}}}
\newcommand{\srule}{\begin{center}{\color{housedivider}\rule{0.5\linewidth}{0.5pt}}\end{center}}

\newlist{clause}{enumerate}{3}
\setlist[clause,1]{label=\textbf{\arabic*.}, ref=\arabic*, leftmargin=2.2em, labelsep=0.6em, itemsep=0.5em, topsep=0.35em}
\setlist[clause,2]{label=\arabic{clausei}.\arabic*, ref=\arabic{clausei}.\arabic*, leftmargin=2.6em, labelsep=0.6em, itemsep=0.35em, topsep=0.3em}
\setlist[clause,3]{label=(\alph*), ref=\arabic{clausei}.\arabic{clauseii}(\alph*), leftmargin=2.0em, labelsep=0.5em, itemsep=0.25em, topsep=0.25em}
\newcommand{\ctitle}[1]{\textbf{\headingfont\color{houseprimary}#1}}

\newcommand{\housetitle}[3]{%
  {\headingfont\fontsize{18}{22}\selectfont\bfseries\color{houseprimary}#1\par}%
  \medskip
  {\color{housemuted}#2\quad|\quad#3\par}%
  \bigskip}

\newenvironment{housenotice}{\begin{quote}\small\itshape}{\end{quote}\medskip}

\newenvironment{housecontrol}
  {\par\small\tabularx{\linewidth}{@{}>{\leavevmode\bfseries\color{housemuted}}p{3.6cm}X@{}}\toprule}
  {\bottomrule\endtabularx\par\medskip}

% The classification sits at the head and foot of every page: header centre, foot left, with
% the page number kept in the centre of the foot. The header gets the height fancyhdr needs,
% and the top margin gives it back, so the text block does not move.
\newcommand{\houseclassificationmark}{}
\newcommand{\houseclassification}[1]{%
  \renewcommand{\houseclassificationmark}{{\small\bfseries\color{houseprimary}#1}}%
  \ifdim\headheight<14pt
    \addtolength{\topmargin}{\headheight}\setlength{\headheight}{14pt}%
    \addtolength{\topmargin}{-\headheight}%
  \fi
  \fancypagestyle{houseclassified}{\houseclassificationstyle}%
  \fancypagestyle{plain}{\houseclassificationstyle}%
  \pagestyle{houseclassified}}
\newcommand{\houseclassificationstyle}{%
  \fancyhf{}%
  \fancyhead[C]{\houseclassificationmark}%
  \fancyfoot[L]{\houseclassificationmark}%
  \fancyfoot[C]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}}
